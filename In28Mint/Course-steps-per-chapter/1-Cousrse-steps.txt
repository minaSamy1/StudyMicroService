					  @FeignClient(name = "currency-exchange", url = "localhost:8000")  // here defining with serviceName and
						public interface CurrencyExchageProxy {
							// get the Signature of Api from the service controller and JUST replace the Object that will return to be [ CurrencyConversion] as it have the same field for Repsonse
							@GetMapping("/currency-exchange/from/{from}/to/{to}")
							public CurrencyConversion getExchange(@PathVariable String from, @PathVariable String to);
						}
						
				3- calling
				     @Autowired
                      private CurrencyExchageProxy proxy;
				      CurrencyConversion currencyConversion = proxy.getExchange(from, to);

      ----------------------  Api Gateway ---------------------------
the Usage of Api-Gateway  ( it's our Entry point )
   1- is to call the All services we have through one url and port localhost:8765
   2- can here define the all layer of authentication and autherization
 
	  
	  -create project : api-gateway
           pom :  EurkaClient  , Reactive Gateway 
		   application.properities : 
		        spring.cloud.gateway.discovery.locator.enabled=true
                spring.cloud.gateway.discovery.locator.lowerCaseServiceId=true

locator.enabled : This enables the Service Discovery Locator feature in Spring Cloud Gateway
                  will automatically discover and route requests to microservices registered in a service registry 
				  
				 http://localhost:8765/CURRENCY-EXCHANGE/currency-exchange/from/USD/to/INR
				 http://localhost:8765/CURRENCY-CONVERSION/currency-conversion/from/USD/to/INR/quantity/10
				 
lowerCaseServiceId=true : ensures that the service IDs used in the routing configuration are converted to lowercase.
               http://localhost:8765/currency-exchange/currency-exchange/from/USD/to/INR


   - can write custome path for each services 
	     
    @Bean
    public RouteLocator getRouter(RouteLocatorBuilder builder) {


        return builder.routes().route(p -> p.path("/get").filters(f -> f.addRequestHeader("MyHeader", "Header").addRequestParameter("Parm", "ParamValue")).uri("http://httpbin.org:80")).
                route(p -> p.path("/currency-exchange/**").uri("lb://CURRENCY-EXCHANGE")).
                route(p -> p.path("/currency-conversion/**").uri("lb://CURRENCY-CONVERSION")).
                route(p -> p.path("/currency-conversionFegin/**").uri("lb://CURRENCY-CONVERSION"))
                .route(p -> p.path("/currency-conversion-new/**")
                        .filters(f -> f.rewritePath(
                                "/currency-conversion-new/(?<segment>.*)",
                                "/currency-conversionFegin/${segment}"))
                        .uri("lb://currency-conversion")).build();

               }
